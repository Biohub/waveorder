"""Batch Renormalization (Ioffe 2017, https://arxiv.org/abs/1702.03275).

Motivated by the amplitude/dynamic-range washout Gav found: extreme-range
FOVs (bright outlier cells) skew a minibatch's statistics, and plain
BatchNorm's implicit "regress to the batch mean" behavior collapses the
output's dynamic range on exactly those FOVs (2026-09 meeting w/ Shalin).
Batch Renorm corrects the normalized activation toward the running
(dataset-level) statistics with a clipped affine term ``(r, d)`` whose
bounds relax from ``(1, 0)`` -- i.e. plain BatchNorm -- to
``(rmax_final, dmax_final)`` over ``warmup_steps``, so early training (when
running stats are unreliable) is unaffected and later training becomes
progressively less minibatch-dependent.

``BatchRenorm2d`` keeps ``nn.BatchNorm2d``'s parameter/buffer names
(``weight``, ``bias``, ``running_mean``, ``running_var``,
``num_batches_tracked``), so ``convert_to_batchrenorm`` can swap it in for
every BatchNorm2d in a module tree -- including inside a pretrained timm
encoder -- while carrying over the existing (e.g. ImageNet-pretrained)
statistics as the starting point.
"""

from __future__ import annotations

import torch
import torch.nn as nn


class BatchRenorm2d(nn.Module):
    def __init__(self, num_features: int, eps: float = 1e-5,
                 momentum: float = 0.1, affine: bool = True,
                 rmax_final: float = 3.0, dmax_final: float = 5.0,
                 warmup_steps: int = 5000):
        super().__init__()
        self.num_features = num_features
        self.eps = eps
        self.momentum = momentum if momentum is not None else 0.1
        self.affine = affine
        self.rmax_final = rmax_final
        self.dmax_final = dmax_final
        self.warmup_steps = warmup_steps

        if affine:
            self.weight = nn.Parameter(torch.ones(num_features))
            self.bias = nn.Parameter(torch.zeros(num_features))
        else:
            self.register_parameter("weight", None)
            self.register_parameter("bias", None)

        self.register_buffer("running_mean", torch.zeros(num_features))
        self.register_buffer("running_var", torch.ones(num_features))
        self.register_buffer("num_batches_tracked", torch.tensor(0, dtype=torch.long))

    def _rmax_dmax(self) -> tuple[float, float]:
        t = min(1.0, float(self.num_batches_tracked.item()) / max(1, self.warmup_steps))
        return 1.0 + (self.rmax_final - 1.0) * t, self.dmax_final * t

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if x.dim() != 4:
            raise ValueError(f"BatchRenorm2d expects (N,C,H,W), got {tuple(x.shape)}")

        if not self.training:
            x_hat = ((x - self.running_mean[None, :, None, None])
                     / torch.sqrt(self.running_var[None, :, None, None] + self.eps))
        else:
            dims = (0, 2, 3)
            batch_mean = x.mean(dim=dims)
            batch_var = x.var(dim=dims, unbiased=False)
            std = torch.sqrt(batch_var + self.eps)
            running_std = torch.sqrt(self.running_var + self.eps)

            rmax, dmax = self._rmax_dmax()
            r = (std.detach() / running_std).clamp(1.0 / rmax, rmax)
            d = ((batch_mean.detach() - self.running_mean) / running_std).clamp(-dmax, dmax)

            x_hat = (x - batch_mean[None, :, None, None]) / std[None, :, None, None]
            x_hat = x_hat * r[None, :, None, None] + d[None, :, None, None]

            with torch.no_grad():
                n = x.numel() / x.size(1)
                unbiased_var = batch_var * n / max(n - 1, 1)
                self.running_mean += self.momentum * (batch_mean - self.running_mean)
                self.running_var += self.momentum * (unbiased_var - self.running_var)
                self.num_batches_tracked += 1

        if self.affine:
            x_hat = x_hat * self.weight[None, :, None, None] + self.bias[None, :, None, None]
        return x_hat

    def extra_repr(self) -> str:
        return (f"{self.num_features}, eps={self.eps}, momentum={self.momentum}, "
                f"rmax_final={self.rmax_final}, dmax_final={self.dmax_final}, "
                f"warmup_steps={self.warmup_steps}")


def convert_to_batchrenorm(module: nn.Module, **br_kwargs) -> nn.Module:
    """Recursively replace every ``nn.BatchNorm2d`` in ``module`` with a
    ``BatchRenorm2d``, in place, carrying over weight/bias/running stats.
    """
    for name, child in module.named_children():
        if isinstance(child, nn.BatchNorm2d):
            br = BatchRenorm2d(child.num_features, eps=child.eps,
                               momentum=child.momentum, affine=child.affine,
                               **br_kwargs)
            if child.affine:
                br.weight.data.copy_(child.weight.data)
                br.bias.data.copy_(child.bias.data)
            br.running_mean.copy_(child.running_mean)
            br.running_var.copy_(child.running_var)
            br.num_batches_tracked.copy_(child.num_batches_tracked)
            setattr(module, name, br)
        else:
            convert_to_batchrenorm(child, **br_kwargs)
    return module
